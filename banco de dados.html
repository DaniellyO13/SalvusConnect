🔧 salvusconnect — banco de dados

mysql, relacionamentos e estrutura das tabelas

modelo do sistema: este documento apresenta as oito tabelas do projeto: usuários, telefones, veículos, serviços, agendamentos, avaliações, antecedentes criminais e recuperação de senha. o tipo de usuário é identificado pela coluna cd_tipo_usuario da tabela tb_usuario: 0 = administrador, 1 = prestador e 3 = cliente.
atenção: o script sql abaixo é uma referência para criar o banco em uma instalação nova. ele não é uma migração para o banco atual. não execute a criação completa sobre um banco com dados existentes; faça um backup e confira as estruturas antes de aplicar mudanças. a instrução de cadastro dos serviços também deve ser executada apenas uma vez para evitar duplicações.

  • usuário → veículos: tb_automovel.tb_usuario_id_usuario.
  • serviço → agendamentos: tb_agendamento.tb_servicos_id_servico.
  • cliente → agendamentos: tb_agendamento.tb_usuario_id_usuario identifica quem solicitou o serviço.
  • prestador → agendamentos: tb_agendamento.tb_prestador_id_usuario referencia tb_usuario.id_usuario e pode ficar nulo antes da atribuição.
  • agendamento → avaliações: tb_avaliacao.tb_agendamento_id_agendamento.
  • usuário → antecedentes: tb_antecedentes.tb_usuario_id_usuario referencia o usuário que enviou o documento.
  • administrador → antecedentes: tb_antecedentes.tb_administrador_id_usuario referencia tb_usuario.id_usuario para identificar quem analisa o documento.
  • usuário → recuperação de senha: tb_recuperacao_senha.tb_usuario_id_usuario vincula o token à conta.
  • tipos de usuário: não há tabelas separadas para administrador, prestador e cliente; os perfis são indicados por cd_tipo_usuario.

💻 script sql — modelo para instalação nova

este script descreve o modelo aprovado. execute a criação completa somente em uma instalação nova e vazia. create table if not exists não atualiza tabelas existentes. os dez serviços são inseridos ao final e não devem ser inseridos novamente se já estiverem cadastrados.

create database if not exists salvusconnect;
use salvusconnect;

-- tabela de usuarios: 0 = administrador, 1 = prestador, 3 = cliente
create table if not exists tb_usuario (
    id_usuario int primary key auto_increment,
    nm_nome varchar(100) not null,
    nr_cpf char(11),
    cnh varchar(20),
    nm_tipo_cnh varchar(3),
    email varchar(150) not null,
    senha varchar(255) not null,
    cd_tipo_usuario int not null,
    ds_sobre_mim text null,
    check (cd_tipo_usuario in (0, 1, 3))
) engine=innodb;

-- tabela de telefones
create table if not exists tb_telefone (
    id_telefone int primary key auto_increment,
    nr_telefone char(11),
    nm_tipo_telefone varchar(45),
    tb_usuario_id_usuario int not null,
    foreign key (tb_usuario_id_usuario)
        references tb_usuario(id_usuario)
) engine=innodb;

-- tabela de automoveis
create table if not exists tb_automovel (
    id_automovel int primary key auto_increment,
    nm_marca varchar(50),
    nm_modelo varchar(50),
    ano_modelo int,
    nm_cor varchar(30),
    nr_placa varchar(10),
    tb_usuario_id_usuario int not null,
    foreign key (tb_usuario_id_usuario)
        references tb_usuario(id_usuario)
) engine=innodb;

-- tabela de servicos
create table if not exists tb_servicos (
    id_servico int primary key auto_increment,
    nm_servico varchar(200),
    ds_descricao varchar(500),
    nr_duracao_minutos int not null default 120
) engine=innodb;

-- tabela de agendamentos
create table if not exists tb_agendamento (
    id_agendamento int primary key auto_increment,
    st_agendamento varchar(20) not null default 'pendente',
    hr_horario datetime,
    ds_local_busca varchar(200),
    ds_local_entrega varchar(200),
    tb_servicos_id_servico int not null,
    tb_usuario_id_usuario int not null,
    tb_prestador_id_usuario int null,
    check (
        st_agendamento in (
            'pendente',
            'aceito',
            'recusado',
            'em_andamento',
            'concluido',
            'cancelado'
        )
    ),
    foreign key (tb_servicos_id_servico)
        references tb_servicos(id_servico),
    foreign key (tb_usuario_id_usuario)
        references tb_usuario(id_usuario),
    constraint fk_agendamento_prestador
        foreign key (tb_prestador_id_usuario)
        references tb_usuario(id_usuario)
) engine=innodb;

-- tabela de avaliacoes
create table if not exists tb_avaliacao (
    id_avaliacao int primary key auto_increment,
    nr_nota int,
    ds_comentario text,
    tb_agendamento_id_agendamento int not null,
    check (nr_nota is null or nr_nota between 1 and 5),
    foreign key (tb_agendamento_id_agendamento)
        references tb_agendamento(id_agendamento)
) engine=innodb;

-- tabela de antecedentes criminais
create table if not exists tb_antecedentes (
    id_antecedente int primary key auto_increment,
    nm_arquivo varchar(255) not null,
    ds_arquivo varchar(500) not null,
    nm_tipo_arquivo varchar(50),
    dt_envio datetime default current_timestamp,
    st_analise varchar(20) not null default 'PENDENTE',
    dt_analise datetime,
    dt_proxima_atualizacao date null,
    ds_observacao varchar(500),
    tb_usuario_id_usuario int not null,
    tb_administrador_id_usuario int,
    check (st_analise in ('PENDENTE', 'APROVADO', 'REPROVADO')),
    foreign key (tb_usuario_id_usuario)
        references tb_usuario(id_usuario),
    foreign key (tb_administrador_id_usuario)
        references tb_usuario(id_usuario)
) engine=innodb;

-- tabela de recuperacao de senha
create table if not exists tb_recuperacao_senha (
    id_recuperacao bigint unsigned primary key auto_increment,
    tb_usuario_id_usuario int not null,
    token_hash char(64)
        character set ascii
        collate ascii_bin not null,
    dt_expiracao datetime not null,
    dt_utilizado datetime null,
    dt_criacao datetime not null default current_timestamp,
    unique key uq_recuperacao_token (token_hash),
    index idx_recuperacao_usuario (tb_usuario_id_usuario),
    index idx_recuperacao_expiracao (dt_expiracao),
    constraint fk_recuperacao_usuario
        foreign key (tb_usuario_id_usuario)
        references tb_usuario(id_usuario)
        on delete cascade
        on update cascade
) engine=innodb;
